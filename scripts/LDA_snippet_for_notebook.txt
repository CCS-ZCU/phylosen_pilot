# CELL 3: Visual comparison PCA vs LDA
from sklearn.decomposition import PCA
import matplotlib.pyplot as plt

# LDA
X_lda, lda_model = compute_lda_projection(X_clean, y_passio)
sil_lda = silhouette_score(X_lda, y_passio)

# PCA
pca = PCA(n_components=2, random_state=42)

print(f"\nImprovement: {(sil_lda - sil_pca):.4f} ({(sil_lda - sil_pca)/abs(sil_pca)*100:.1f}%)")

axes[1].scatter(X_lda[:, 0], X_lda[:, 1], c=y_passio, cmap='tab10', s=20, alpha=0.6)

plt.tight_layout()
plt.show()
